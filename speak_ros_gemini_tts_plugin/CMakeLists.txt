cmake_minimum_required(VERSION 3.14)
project(speak_ros_gemini_tts_plugin)

if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
find_package(ament_cmake_auto REQUIRED)
find_package(cpprestsdk REQUIRED)
ament_auto_find_build_dependencies()
ament_auto_add_library(${PROJECT_NAME} SHARED src/${PROJECT_NAME}.cpp)
target_link_libraries(${PROJECT_NAME} cpprestsdk::cpprest)
pluginlib_export_plugin_description_file(speak_ros plugins.xml)

if(BUILD_TESTING)
  ament_auto_add_executable(audio_events_test test/audio_events_test.cpp)
  target_link_libraries(audio_events_test cpprestsdk::cpprest)
  add_test(NAME audio_events_contract COMMAND $<TARGET_FILE:audio_events_test>)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  ament_auto_add_executable(streaming_client test/streaming_client.cpp)
  target_link_libraries(streaming_client ${PROJECT_NAME})
  add_test(NAME streaming_http COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_streaming_http.py $<TARGET_FILE:streaming_client>)
  set_tests_properties(streaming_http PROPERTIES TIMEOUT 90)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()
ament_auto_package(USE_SCOPED_HEADER_INSTALL_DIR)
